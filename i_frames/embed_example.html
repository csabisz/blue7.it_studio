AI Image Modal - Iframe Embed Example

This page demonstrates how to embed the AI Image Generation modal in any website using an iframe. The modal communicates with the parent page via postMessage events.

Integration Code

Here's how to integrate this modal into your website:

// 1. Create the iframe const iframe = document.createElement('iframe'); iframe.src = 'https://cseven.eu/studio/i_frames/ai_image_modal.php?orf_id=123&token=YOUR_TOKEN'; iframe.style.width = '100%'; iframe.style.height = '800px'; iframe.style.border = 'none'; document.body.appendChild(iframe); // 2. Listen for events from the iframe window.addEventListener('message', function(event) { // Verify the message is from our modal if (event.data.type !== 'ai-modal-event') return; const { event: eventName, data } = event.data; switch (eventName) { case 'ready': console.log('Modal is ready', data); break; case 'imageGenerated': console.log('New image generated!', data); // data contains: id, image_url, thumbnail_url, model, quality break; case 'imageSaved': console.log('Image saved to task', data); // data contains: orf_ai_id, saved_orf_id break; case 'error': console.error('Error occurred', data); // data contains: message, code break; case 'close': console.log('User closed modal'); // Remove iframe or hide container iframe.remove(); break; } });
